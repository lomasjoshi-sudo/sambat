create table public.notes ( id uuid primary key default gen_random_uuid(), user_id uuid references auth.users(id) on delete cascade not null, date_key text not null, -- format: "YYYY-MM-DD" (AD date) content text not null, created_at timestamptz default now(), updated_at timestamptz default now(), unique(user_id, date_key) -- one note per user per day ); alter table public.notes enable row level security; create policy "Users can view their own notes" on public.notes for select to authenticated using ((select auth.uid()) = user_id); create policy "Users can insert their own notes" on public.notes for insert to authenticated with check ((select auth.uid()) = user_id); create policy "Users can update their own notes" on public.notes for update to authenticated using ((select auth.uid()) = user_id); create policy "Users can delete their own notes" on public.notes for delete to authenticated using ((select auth.uid()) = user_id);
